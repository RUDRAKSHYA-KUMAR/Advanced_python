    vowels = "aeiouAEIOU"
    vcount = 0
    ccount=0
    for char in text:
        if char in vowels:
            vcount += 1
        else:
            ccount+=1
    print("Total no of vowel are: ",vcount)

12. Write a program: - Accept list of numbers - Remove duplicates 
- Sort it without using sort() 
ANSWER-

numbers = list(map(int, input("Enter numbers separated by space: ").split()))
unique_numbers = []
for num in numbers:
    if num not in unique_numbers:
        unique_numbers.append(num)
n = len(unique_numbers)

for i in range(n):
    for j in range(0, n - i - 1):
        if unique_numbers[j] > unique_numbers[j + 1]:
            # Swap
            temp = unique_numbers[j]
            unique_numbers[j] = unique_numbers[j + 1]
            unique_numbers[j + 1] = temp

13. Write a program using lambda + map + filter: - Square only even numbers from list 
ANSWER-

numbers = list(map(int, input("Enter numbers separated by space: ").split()))

result = list(
    map(lambda x: x**2, 
        filter(lambda x: x % 2 == 0, numbers))
)
print("Squared even numbers:", result)

20. Build mini project: 
STUDENT MANAGEMENT SYSTEM 
Features: - Add student - View student - Delete student - Store data in file or database 
ANSWER-
students = {}

while True:
    print("\n--- STUDENT MANAGEMENT SYSTEM ---")
    print("1. Add Student")
    print("2. View Students")
    print("3. Delete Student")
    print("4. Exit")

    choice = input("Enter choice: ")
    if choice == "1":
        sid = input("Enter student ID: ")
        name = input("Enter name: ")
        age = input("Enter age: ")
        course = input("Enter course: ")

        students[sid] = {
            "name": name,
            "age": age,
            "course": course
        }

        print(" Student added!")

    
    elif choice == "2":
        if not students:
            print("No students found")
        else:
            for sid, data in students.items():
                print(f"ID: {sid} | Name: {data['name']} | Age: {data['age']} | Course: {data['course']}")

    elif choice == "3":
        sid = input("Enter student ID to delete: ")

        if sid in students:
            del students[sid]
            print(" Student deleted!")
        else:
            print(" Student not found")


    elif choice == "4":
        print(" Exiting...")
        break

    else:
        print("Invalid choice")
